5. Pass empno as a in parameter to the procedure. If empno is
found in the table then, display corresponding salary out of
the procedure. If empno is not found in the table then, display
0 out of the procedure. 

DELIMITER //

CREATE PROCEDURE get_emp_salary(
    IN in_empno INT,
    OUT out_sal DECIMAL(10,2)
)
BEGIN
    DECLARE emp_count INT;

    SELECT COUNT(*) INTO emp_count 
    FROM emp 
    WHERE empno = in_empno;

    IF emp_count > 0 THEN
        SELECT sal INTO out_sal 
        FROM emp 
        WHERE empno = in_empno;
    ELSE
        SET out_sal = 0;
    END IF;
END //

DELIMITER ;

CALL get_emp_salary(7499, @salary1);
SELECT @salary1 AS "Result Salary";

2. Write a function to generate an email. Call the function in
select clause. email is generated as follows
​​​​​ first 2 letters of name + last 2 letters of job @knowit.com

DELIMITER //

CREATE FUNCTION generate_email(in_name VARCHAR(50), in_job VARCHAR(50))
RETURNS VARCHAR(100)
DETERMINISTIC
BEGIN
    DECLARE first_two VARCHAR(2);
    DECLARE last_two VARCHAR(2);
    
    SET first_two = LEFT(in_name, 2);
    SET last_two = RIGHT(in_job, 2);
    
    RETURN LOWER(CONCAT(first_two, last_two, '@knowit.com'));
END //

DELIMITER ;

SELECT ename, job, generate_email(ename, job) AS "Generated Email" 
FROM emp;


3. Write a function which returns a factorial of a number to the
procedure 'Facto'. Check if number is positive. Print the
factorial of a number outside the procedure. Pass number as a
in parameter to the procedure. Call function to calculate
factorial of a number in the procedure. Function should return
factorial to the procedure and procedure should display number
and the factorial out of procedure.

DELIMITER //

CREATE FUNCTION calc_factorial(num INT)
RETURNS BIGINT
DETERMINISTIC
BEGIN
    DECLARE fact BIGINT DEFAULT 1;
    DECLARE i INT DEFAULT 1;
    
    WHILE i <= num DO
        SET fact = fact * i;
        SET i = i + 1;
    END WHILE;
    
    RETURN fact;
END //
DELIMITER ;

DELIMITER //

CREATE PROCEDURE Facto(
    IN in_num INT,
    OUT out_num INT,
    OUT out_fact BIGINT
)
BEGIN
    SET out_num = in_num;
    
    IF in_num > 0 THEN
        SET out_fact = calc_factorial(in_num);
    ELSE
        SET out_fact = 0;
    END IF;
END //

DELIMITER ;

CALL Facto(5, @num_out, @fact_out);
SELECT @num_out AS "Number", @fact_out AS "Factorial";


4. Pass empno as a parameter to the procedure and display
'Award' as a remark   if experience of employee is more
than 44 years else display null for that employee.
Write a function which will calculate and return
experience (no of years of services) to the procedure.
Expected Output:
ename deptno experience remark
----------------------------------
Amit     10        44    null
Seema    20         45     Award

DELIMITER //

CREATE FUNCTION get_experience(in_hiredate DATE)
RETURNS INT
DETERMINISTIC
BEGIN
    
    RETURN TIMESTAMPDIFF(YEAR, in_hiredate, CURDATE());
END //

DELIMITER ;

DELIMITER //

CREATE PROCEDURE check_employee_award(IN in_empno INT)
BEGIN
    SELECT 
        ename, 
        deptno, 
        get_experience(hiredate) AS experience,
        CASE 
            WHEN get_experience(hiredate) > 44 THEN 'Award'
            ELSE NULL 
        END AS remark
    FROM emp
    WHERE empno = in_empno;
END //

DELIMITER ;

CALL check_employee_award(7499);

SELECT 
    e.ename, 
    e.deptno, 
    d.dname, 
    d.loc,
    MAX(e.sal) OVER(PARTITION BY e.deptno) AS "Maximum Sal",
    MIN(e.sal) OVER(PARTITION BY e.deptno) AS "Minimum Sal",

JOIN dept d ON e.deptno = d.deptno;

WITH RankedEmployees AS (
SELECT ename, deptno, job, sal, AVG(sal) OVER (PARTITION BY job) AS avg_sal,

FROM emp

FROM RankedEmployees
WHERE rn = 1;
