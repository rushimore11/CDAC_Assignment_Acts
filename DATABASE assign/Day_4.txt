-Display total spendings per customer.
select customer_id , sum(amount) from orders group by customer_id;

-Display Top 3 customers as per their spendings
select customer_id, sum(amount) as Total_spendings from orders group by customer_id order by Total_spendings desc limit 3 ;

-Generate Monthly sales report
SELECT 
    YEAR(order_date) AS Year, 
    MONTHNAME(order_date) AS Month, 
    COUNT(order_id) AS Total_Orders, 
    SUM(amount) AS Total_Sales_Amount
FROM 
    Orders
GROUP BY 
    YEAR(order_date), 
    MONTH(order_date),
    MONTHNAME(order_date)
ORDER BY 
    Year DESC, 
    MONTH(order_date) DESC;



-Display customer_id, name and city of customers who ordered
 more than 3 times
 SELECT 
    c.customer_id, 
    c.name, 
    c.city
FROM 
    Customers c
JOIN 
    Orders o ON c.customer_id = o.customer_id
GROUP BY 
    c.customer_id, 
    c.name, 
    c.city
HAVING 
    COUNT(o.order_id) > 3;

 
-Find customers who spent more than ₹2000 as a total amount.
SELECT customer_id, SUM(amount) AS total_spending
FROM Orders
GROUP BY customer_id
HAVING SUM(amount) > 2000;


-Count the number of orders placed by each customer
SELECT 
    customer_id, 
    COUNT(order_id) AS total_orders
FROM 
    Orders
GROUP BY 
    customer_id
ORDER BY

-List customers who placed orders in February
SELECT DISTINCT 
    c.customer_id, 
    c.name, 
    c.city

-Find the second highest order amount
SELECT DISTINCT amount

-Identify the customer with the highest total spending
SELECT 
    c.customer_id, 
    c.name, 
    SUM(o.amount) AS total_spent
FROM 
    Customers c
JOIN 
    Orders o ON c.customer_id = o.customer_id
GROUP BY 
    c.customer_id, 
    c.name
ORDER BY 
    total_spent DESC
LIMIT 1;


-Rank customers based on their total spending
SELECT 
    c.customer_id, 
    c.name, 
    SUM(o.amount) AS total_spent,
    DENSE_RANK() OVER (ORDER BY SUM(o.amount) DESC) AS customer_rank
FROM 
    Customers c
JOIN 
    Orders o ON c.customer_id = o.customer_id
GROUP BY 
    c.customer_id, 
    c.name;


-Identify duplicate order amounts
SELECT amount, COUNT(amount) AS occurrence
FROM Orders

HAVING COUNT(amount) > 1;


-Find orders with amount greater than the average order amount
